WITH params AS (
  SELECT '{{Select_Stats_Month.selectedOptionValue}}' AS selected_month
),
-- חודשים היסטוריים מהטבלה הידנית
historical AS (
  SELECT month AS חודש, meetings AS פגישות, closings AS סגירות,
         bonus_meetings AS בונוס_פגישות, bonus_closings AS בונוס_סגירות, total_bonus AS סה_כ_בונוס
  FROM lidor_bonus
  WHERE (SELECT selected_month FROM params) = ''
     OR month = (SELECT selected_month FROM params)
),
-- חודשים שלא בטבלה הידנית — מחושב מהנתונים
dynamic AS (
  SELECT
    TO_CHAR(DATE_TRUNC('month', update_date), 'MM/YYYY') AS חודש,
    COUNT(DISTINCT CASE WHEN meeting_held = 'כן' THEN lead_id END) AS פגישות,
    COUNT(DISTINCT CASE WHEN deal_closed = 'כן' THEN lead_id END) AS סגירות,
    COUNT(DISTINCT CASE WHEN meeting_held = 'כן' THEN lead_id END) * 20 AS בונוס_פגישות,
    COUNT(DISTINCT CASE WHEN deal_closed = 'כן' THEN lead_id END) * 50 AS בונוס_סגירות,
    (COUNT(DISTINCT CASE WHEN meeting_held = 'כן' THEN lead_id END) * 20) +
    (COUNT(DISTINCT CASE WHEN deal_closed = 'כן' THEN lead_id END) * 50) AS סה_כ_בונוס
  FROM lidor_leads_view
  WHERE touched_by_lidor = TRUE
    AND update_date IS NOT NULL
    AND NOT EXISTS (
      SELECT 1 FROM lidor_bonus
      WHERE month = TO_CHAR(DATE_TRUNC('month', update_date), 'MM/YYYY')
    )
    AND (
      (SELECT selected_month FROM params) = ''
      OR TO_CHAR(DATE_TRUNC('month', update_date), 'MM/YYYY') = (SELECT selected_month FROM params)
    )
  GROUP BY TO_CHAR(DATE_TRUNC('month', update_date), 'MM/YYYY')
)
SELECT * FROM historical
UNION ALL
SELECT * FROM dynamic
ORDER BY חודש DESC;
