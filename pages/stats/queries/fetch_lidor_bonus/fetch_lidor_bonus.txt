WITH params AS (
  SELECT CASE
    WHEN '{{Select_Stats_Month.selectedOptionValue}}' = '' THEN CURRENT_DATE
    WHEN TO_CHAR(CURRENT_DATE, 'MM/YYYY') = '{{Select_Stats_Month.selectedOptionValue}}' THEN CURRENT_DATE
    ELSE (TO_DATE('{{Select_Stats_Month.selectedOptionValue}}', 'MM/YYYY') + INTERVAL '1 month - 1 day')::date
  END AS ref_date
)
SELECT
  TO_CHAR(COALESCE(close_date, update_date), 'MM/YYYY') AS חודש,
  COUNT(DISTINCT CASE WHEN lead_passed = TRUE THEN lead_id END) AS לידים_עברו,
  COUNT(DISTINCT CASE WHEN deal_closed = 'כן' THEN lead_id END) AS סגירות,
  COUNT(DISTINCT CASE WHEN lead_passed = TRUE THEN lead_id END) * 20 AS בונוס_לידים,
  COUNT(DISTINCT CASE WHEN deal_closed = 'כן' THEN lead_id END) * 50 AS בונוס_סגירות,
  (COUNT(DISTINCT CASE WHEN lead_passed = TRUE THEN lead_id END) * 20) +
  (COUNT(DISTINCT CASE WHEN deal_closed = 'כן' THEN lead_id END) * 50) AS סה_כ_בונוס
FROM lidor_leads_view
WHERE touched_by_lidor = TRUE
AND DATE_TRUNC('month', COALESCE(close_date, update_date)) = DATE_TRUNC('month', (SELECT ref_date FROM params))
GROUP BY TO_CHAR(COALESCE(close_date, update_date), 'MM/YYYY')
ORDER BY MIN(COALESCE(close_date, update_date)) DESC;
