WITH params AS (
  SELECT '{{Select_Stats_Month.selectedOptionValue}}' AS selected_month
)
SELECT month AS חודש, meetings AS פגישות, closings AS סגירות, 
       bonus_meetings AS בונוס_פגישות, bonus_closings AS בונוס_סגירות, total_bonus AS סה_כ_בונוס
FROM lidor_bonus
WHERE (SELECT selected_month FROM params) = ''
   OR month = (SELECT selected_month FROM params)

UNION ALL

SELECT
  TO_CHAR(DATE_TRUNC('month', CURRENT_DATE), 'MM/YYYY'),
  COUNT(DISTINCT CASE WHEN meeting_held = 'כן' THEN lead_id END),
  COUNT(DISTINCT CASE WHEN close_date IS NOT NULL AND DATE_TRUNC('month', close_date) = DATE_TRUNC('month', CURRENT_DATE) THEN lead_id END),
  COUNT(DISTINCT CASE WHEN meeting_held = 'כן' THEN lead_id END) * 20,
  COUNT(DISTINCT CASE WHEN close_date IS NOT NULL AND DATE_TRUNC('month', close_date) = DATE_TRUNC('month', CURRENT_DATE) THEN lead_id END) * 50,
  (COUNT(DISTINCT CASE WHEN meeting_held = 'כן' THEN lead_id END) * 20) +
  (COUNT(DISTINCT CASE WHEN close_date IS NOT NULL AND DATE_TRUNC('month', close_date) = DATE_TRUNC('month', CURRENT_DATE) THEN lead_id END) * 50)
FROM lidor_leads_view
WHERE touched_by_lidor = TRUE
AND (
  (SELECT selected_month FROM params) = ''
  OR (SELECT selected_month FROM params) = TO_CHAR(DATE_TRUNC('month', CURRENT_DATE), 'MM/YYYY')
)
AND NOT EXISTS (
  SELECT 1 FROM lidor_bonus 
  WHERE month = TO_CHAR(DATE_TRUNC('month', CURRENT_DATE), 'MM/YYYY')
)

ORDER BY חודש DESC;
