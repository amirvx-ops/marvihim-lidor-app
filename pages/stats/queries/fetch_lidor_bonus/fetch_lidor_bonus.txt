WITH params AS (
  SELECT CASE
    WHEN '{{Select_Stats_Month.selectedOptionValue}}' = '' THEN CURRENT_DATE
    WHEN TO_CHAR(CURRENT_DATE, 'MM/YYYY') = '{{Select_Stats_Month.selectedOptionValue}}' THEN CURRENT_DATE
    ELSE (TO_DATE('{{Select_Stats_Month.selectedOptionValue}}', 'MM/YYYY') + INTERVAL '1 month - 1 day')::date
  END AS ref_date
)
SELECT
  TO_CHAR((SELECT ref_date FROM params), 'MM/YYYY') AS חודש,
  COUNT(DISTINCT CASE WHEN meeting_held = 'כן'
    AND DATE_TRUNC('month', update_date) = DATE_TRUNC('month', (SELECT ref_date FROM params))
    THEN lead_id END) AS פגישות,
  COUNT(DISTINCT CASE WHEN close_date IS NOT NULL
    AND DATE_TRUNC('month', close_date) = DATE_TRUNC('month', (SELECT ref_date FROM params))
    THEN lead_id END) AS סגירות,
  COUNT(DISTINCT CASE WHEN meeting_held = 'כן'
    AND DATE_TRUNC('month', update_date) = DATE_TRUNC('month', (SELECT ref_date FROM params))
    THEN lead_id END) * 20 AS בונוס_פגישות,
  COUNT(DISTINCT CASE WHEN close_date IS NOT NULL
    AND DATE_TRUNC('month', close_date) = DATE_TRUNC('month', (SELECT ref_date FROM params))
    THEN lead_id END) * 50 AS בונוס_סגירות,
  (COUNT(DISTINCT CASE WHEN meeting_held = 'כן'
    AND DATE_TRUNC('month', update_date) = DATE_TRUNC('month', (SELECT ref_date FROM params))
    THEN lead_id END) * 20) +
  (COUNT(DISTINCT CASE WHEN close_date IS NOT NULL
    AND DATE_TRUNC('month', close_date) = DATE_TRUNC('month', (SELECT ref_date FROM params))
    THEN lead_id END) * 50) AS סה_כ_בונוס
FROM lidor_leads_view
WHERE touched_by_lidor = TRUE;
