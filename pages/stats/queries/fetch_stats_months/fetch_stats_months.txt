SELECT month_label FROM (
  SELECT TO_CHAR(created_at, 'MM/YYYY') AS month_label FROM leads WHERE created_at IS NOT NULL
  UNION
  SELECT TO_CHAR(update_date, 'MM/YYYY') FROM lidor_leads_view WHERE touched_by_lidor = TRUE AND update_date IS NOT NULL
) x
ORDER BY TO_DATE(month_label, 'MM/YYYY') DESC;
